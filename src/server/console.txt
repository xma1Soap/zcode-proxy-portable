  <script>
  (function () {
    var SERVER_KEY = __ZCODE_PROXY_KEY_JSON__;
    var toastTimer;
    function $(id) { return document.getElementById(id); }
    function toast(msg) {
      $("toast").textContent = msg;
      $("toast").classList.add("show");
      clearTimeout(toastTimer);
      toastTimer = setTimeout(function () { $("toast").classList.remove("show"); }, 2200);
    }
    function origin() { return location.origin; }
    function fmtMs(ms) {
      if (ms == null) return "—";
      var s = Math.floor(ms / 1000);
      if (s < 60) return s + "s";
      if (s < 3600) return Math.floor(s / 60) + "m " + (s % 60) + "s";
      return Math.floor(s / 3600) + "h " + Math.floor((s % 3600) / 60) + "m";
    }
    function setPulse(kind) {
      $("pulse").className = "pulse " + (kind || "");
    }
    var probeModel = "glm-5.3";
    function setProbeModel(id) {
      probeModel = id;
      $("probe-dd-label").textContent = id;
      var menu = $("probe-dd-menu");
      Array.prototype.forEach.call(menu.children, function (el) {
        el.classList.toggle("active", el.getAttribute("data-id") === id);
      });
    }
    function fillModels(list) {
      var box = $("box-models");
      var menu = $("probe-dd-menu");
      var keep = probeModel;
      var ids = list || [];
      var same = menu.children.length === ids.length;
      if (same) {
        for (var i = 0; i < ids.length; i++) {
          if (menu.children[i].getAttribute("data-id") !== ids[i]) { same = false; break; }
        }
      }
      if (same) return;
      box.innerHTML = "";
      menu.innerHTML = "";
      ids.forEach(function (id) {
        var c = document.createElement("span");
        c.className = "chip"; c.textContent = id; box.appendChild(c);
        var o = document.createElement("button");
        o.type = "button"; o.className = "dd-item"; o.setAttribute("data-id", id); o.textContent = id;
        o.onclick = function (ev) {
          ev.stopPropagation();
          setProbeModel(id);
          $("probe-dd").classList.remove("open");
        };
        menu.appendChild(o);
      });
      if (keep && ids.indexOf(keep) >= 0) setProbeModel(keep);
      else if (ids.indexOf("glm-5.3") >= 0) setProbeModel("glm-5.3");
      else if (ids[0]) setProbeModel(ids[0]);
    }
    function renderAuth(a) {
      if (!a) return;
      $("box-login").textContent = a.loggedIn ? "已登录" : "未登录";
      $("box-user").textContent = (a.account && a.account.userId) || "—";
      $("box-upkey").textContent = (a.account && a.account.apiKeyPreview) || "—";
      $("box-jwt").textContent = a.account && a.account.hasJwt ? "已就绪 (start-plan)" : "无";
      $("box-plan").textContent = (a.provider || "—") + " / " + (a.plan || "—");
      if (a.pending) {
        setPulse("wait");
        $("hero-state").textContent = "等待 " + a.pending.provider + " 授权完成";
        $("hero-title").textContent = "正在登录";
      } else if (a.loggedIn && a.live) {
        setPulse("on");
        $("hero-state").textContent = "服务在线 · 上游已登录";
        $("hero-title").textContent = (a.plan || "ready");
      } else if (a.loggedIn) {
        setPulse("wait");
        $("hero-state").textContent = "已有凭据，但当前模式未使用";
        $("hero-title").textContent = a.plan || "logged in";
      } else {
        setPulse("off");
        $("hero-state").textContent = "服务在线 · 尚未登录上游";
        $("hero-title").textContent = "未登录";
      }
      $("login-hint").textContent = a.lastError
        ? ("上次登录失败：" + a.lastError)
        : "点登录会直接打开智谱 / Z.AI 授权页，不需要先填 API key。";
    }
    function renderStatus(s) {
      if (!s) return;
      $("box-listen").textContent = s.listen || (location.host);
      $("box-reqs").textContent = String(s.requestCount ?? 0);
      $("box-up").textContent = fmtMs(s.uptimeMs);
      if (s.lastRequest) {
        var lr = s.lastRequest;
        $("box-last").textContent = lr.model + " · " + lr.status + " · " + lr.ttfbMs + "ms";
      } else {
        $("box-last").textContent = "还没有请求";
      }
      if (s.models) fillModels(s.models);
      $("hero-line").textContent = (s.listen || location.host) + " · " + (s.provider || "") + " · " + (s.plan || "");
    }
    function refresh() {
      Promise.all([
        fetch("/auth/status").then(function (r) { return r.json(); }).catch(function () { return null; }),
        fetch("/status").then(function (r) { return r.json(); }).catch(function () { return null; })
      ]).then(function (pair) {
        renderAuth(pair[0]);
        renderStatus(pair[1]);
      });
    }
    function startLogin(provider) {
      fetch("/auth/login", {
        method: "POST",
        headers: { "content-type": "application/json" },
        body: JSON.stringify({ provider: provider })
      }).then(function (r) { return r.json(); }).then(function (j) {
        if (!j.authorizeUrl) { toast((j.error && j.error.message) || "无法开始登录"); return; }
        var w = window.open(j.authorizeUrl, "zcode-oauth", "width=520,height=760");
        if (!w) location.href = j.authorizeUrl;
        toast("已打开登录页");
        refresh();
      }).catch(function (e) { toast(String(e)); });
    }
    function logout() {
      fetch("/auth/logout", { method: "POST", headers: { "content-type": "application/json" }, body: "{}" })
        .then(function () { toast("已退出"); refresh(); });
    }
    function importZcode() {
      var headers = { "content-type": "application/json" };
      if (SERVER_KEY) headers.authorization = "Bearer " + SERVER_KEY;
      fetch("/auth/import-zcode", { method: "POST", headers: headers, body: "{}" })
        .then(function (r) { return r.json().then(function (j) { return { ok: r.ok, j: j }; }); })
        .then(function (x) {
          if (!x.ok) { toast((x.j && x.j.error && x.j.error.message) || "ZCode 导入失败"); return; }
          toast("已从 ZCode 导入");
          refresh();
        })
        .catch(function (e) { toast(String(e)); });
    }
    function probe() {
      var model = probeModel || "glm-5.3";
      $("probe-out").textContent = "请求 " + model + " …";
      var headers = { "content-type": "application/json" };
      if (SERVER_KEY) headers.authorization = "Bearer " + SERVER_KEY;
      var t0 = Date.now();
      fetch("/v1/chat/completions", {
        method: "POST",
        headers: headers,
        body: JSON.stringify({
          model: model,
          messages: [{ role: "user", content: "Reply with exactly: ok" }],
          max_tokens: 8,
          stream: false
        })
      }).then(function (r) { return r.text().then(function (t) { return { r: r, t: t }; }); })
        .then(function (x) {
          var ms = Date.now() - t0;
          try {
            var j = JSON.parse(x.t);
            var text = j.choices && j.choices[0] && j.choices[0].message && j.choices[0].message.content;
            $("probe-out").textContent = x.r.status + " · " + ms + "ms · " + (text || x.t.slice(0, 180));
          } catch (e) {
            $("probe-out").textContent = x.r.status + " · " + ms + "ms · " + x.t.slice(0, 180);
          }
          refresh();
        }).catch(function (e) { $("probe-out").textContent = String(e); });
    }
    $("btn-login-bm").onclick = function () { startLogin("bigmodel"); };
    $("btn-login-zai").onclick = function () { startLogin("zai"); };
    function downloadBundle(bundle, name) {
      var blob = new Blob([JSON.stringify(bundle, null, 2)], { type: "application/json" });
      var a = document.createElement("a");
      a.href = URL.createObjectURL(blob);
      a.download = name || "zcode-proxy-credential.json";
      document.body.appendChild(a); a.click(); a.remove();
      URL.revokeObjectURL(a.href);
    }
    function exportCred() {
      var headers = { "content-type": "application/json" };
      if (SERVER_KEY) headers.authorization = "Bearer " + SERVER_KEY;
      fetch("/auth/export-credential", { method: "POST", headers: headers, body: "{}" })
        .then(function (r) { return r.json().then(function (j) { return { ok: r.ok, j: j }; }); })
        .then(function (x) {
          if (!x.ok || !x.j.bundle) { toast((x.j && x.j.error && x.j.error.message) || "导出失败"); return; }
          downloadBundle(x.j.bundle, "zcode-proxy-credential.json");
          toast("已导出解密凭证");
        })
        .catch(function (e) { toast(String(e)); });
    }
    function importCredFile() { $("cred-file").click(); }
    $("cred-file").onchange = function () {
      var f = this.files && this.files[0];
      this.value = "";
      if (!f) return;
      var reader = new FileReader();
      reader.onload = function () {
        var parsed;
        try { parsed = JSON.parse(String(reader.result || "")); }
        catch (e) { toast("JSON 无效"); return; }
        var headers = { "content-type": "application/json" };
        if (SERVER_KEY) headers.authorization = "Bearer " + SERVER_KEY;
        fetch("/auth/import-credential", {
          method: "POST",
          headers: headers,
          body: JSON.stringify({ bundle: parsed })
        }).then(function (r) { return r.json().then(function (j) { return { ok: r.ok, j: j }; }); })
          .then(function (x) {
            if (!x.ok) { toast((x.j && x.j.error && x.j.error.message) || "导入失败"); return; }
            toast("已导入解密凭证");
            refresh();
          }).catch(function (e) { toast(String(e)); });
      };
      reader.readAsText(f);
    };
    $("btn-import-zcode").onclick = importZcode;
    $("btn-export-cred").onclick = exportCred;
    $("btn-import-cred").onclick = importCredFile;
    $("btn-logout").onclick = logout;
    $("btn-probe").onclick = probe;
    $("probe-dd-btn").onclick = function (ev) {
      ev.stopPropagation();
      $("probe-dd").classList.toggle("open");
    };
    document.addEventListener("click", function () {
      $("probe-dd").classList.remove("open");
    });
    $("box-base").textContent = origin() + "/v1";
    $("box-ant").textContent = origin();
    $("box-web").textContent = origin() + "/webui";
    $("box-key").textContent = SERVER_KEY ? (SERVER_KEY.slice(0, 6) + "…" + SERVER_KEY.slice(-4)) : "(未设置)";
    if (window !== window.top) {
      var mgr = $("link-manager");
      if (mgr) mgr.style.display = "none";
    }
    window.addEventListener("message", function (ev) {
      if (ev.data && ev.data.__zcodeOauth === true) { toast(ev.data.ok ? "登录成功" : "登录失败"); refresh(); }
    });
    refresh();
    setInterval(refresh, 3000);
  })();
  </script>
